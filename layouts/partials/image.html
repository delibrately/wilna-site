{{- $path := strings.TrimPrefix "/" .src -}} {{- $resource := resources.Get $path -}} {{- $loading := .loading | default "lazy" -}} {{- if $resource -}} {{- /* Keep existing original-image URLs available alongside optimized variants. */ -}} {{- $originalURL := $resource.RelPermalink -}} {{- $max := int (.width | default 1440) -}} {{- $widths := slice 480 800 1200 1600 1920 -}} {{- $sources := slice -}} {{- range $widths -}} {{- if and (le . $max) (le . $resource.Width) -}} {{- $variant := $resource.Resize (printf "%dx webp q82" .) -}} {{- $sources = $sources | append (printf "%s %dw" $variant.RelPermalink $variant.Width) -}} {{- end -}} {{- end -}} {{- $image := $resource.Resize (printf "%dx webp q82" (math.Min $max $resource.Width | int)) -}} {{- $sources = $sources | append (printf "%s %dw" $image.RelPermalink $image.Width) | uniq -}} {{- else -}} {{- $config := images.Config (printf "static/%s" $path) -}} {{- end -}}
